### 3.2 `eslint.config.js`（ESLint 9 flat config）

```js
import js from '@eslint/js'
import pluginVue from 'eslint-plugin-vue'
import prettier from 'eslint-config-prettier'
import globals from 'globals'

export default [
  {
    ignores: ['dist/**', 'node_modules/**', '*.min.js']
  },

  js.configs.recommended,

  // vue3-recommended：含 no-mutating-props / no-v-html 等关键规则
  ...pluginVue.configs['flat/recommended'],

  // 必须放最后：关掉所有与 Prettier 冲突的格式规则
  prettier,

  {
    languageOptions: {
      ecmaVersion: 'latest',
      sourceType: 'module',
      globals: { ...globals.browser }
    },
    rules: {
      // ---- 真实缺陷类：error ----
      eqeqeq: ['error', 'always', { null: 'ignore' }],
      'no-unused-vars': ['error', { argsIgnorePattern: '^_', varsIgnorePattern: '^_' }],
      'no-undef': 'error',
      'vue/no-mutating-props': 'error',
      'vue/no-dupe-keys': 'error',
      'vue/require-v-for-key': 'error',

      // ---- 安全类：warn（本项目模板里无 v-html，但防未来引入）----
      'vue/no-v-html': 'warn',

      // ---- 可维护性：warn ----
      'no-console': ['warn', { allow: ['warn', 'error'] }],
      'no-debugger': 'error',
      'vue/component-name-in-template-casing': ['warn', 'PascalCase'],
      'vue/require-default-prop': 'warn',

      // ---- 本项目约定：单词组件名（Login / Dashboard）合法 ----
      'vue/multi-word-component-names': 'off',
      // 现有代码用 2 空格缩进，交给 Prettier，ESLint 不判
      'vue/max-attributes-per-line': 'off',
      'vue/singleline-html-element-content-newline': 'off'
    }
  }
]
```

> **注意**：首次运行 `npm run lint` 一定会报一批存量问题。**不要一次全修**——先 `--max-warnings 9999` 让 CI 跑通，然后每周清一批，避免"为了过 lint 提一个 3000 行的 PR"。

## 四、后端：Spotless（推荐）+ Checkstyle（可选）

### 4.1 为什么选 Spotless 而不是 Checkstyle

| | Spotless | Checkstyle |
| --- | --- | --- |
| 工作方式 | **自动改**（`spotless:apply`） | 只报错，人工改 |
| 团队争论 | 零（格式没有讨论空间） | 高（每条规则都要吵） |
| 存量迁移 | 一次 `apply` 全解决 | 逐条修，可能上千条 |

**结论**：格式与 import 用 Spotless；命名、空块等语义检查用 Checkstyle（可选，见 4.3）。

### 4.2 `pom.xml` 加 Spotless

```xml
<plugin>
    <groupId>com.diffplug.spotless</groupId>
    <artifactId>spotless-maven-plugin</artifactId>
    <version>2.43.0</version>
    <configuration>
        <java>
            <!-- AOSP = 4 空格缩进，与现有代码一致；google 风格是 2 空格，会全量重排 -->
            <googleJavaFormat>
                <version>1.22.0</version>
                <style>AOSP</style>
            </googleJavaFormat>
            <removeUnusedImports/>
            <formatAnnotations/>
            <trimTrailingWhitespace/>
            <endWithNewline/>
        </java>
    </configuration>
    <executions>
        <execution>
            <id>spotless-check</id>
            <phase>validate</phase>
            <goals><goal>check</goal></goals>
        </execution>
    </executions>
</plugin>
```

使用：

```bash
cd backend
mvn spotless:apply    # 格式化全项目（会改动文件）
mvn spotless:check    # CI 用，只检查不改
```

> **重要**：首次 `spotless:apply` 会产生大面积 diff。必须**单独一个 `chore/format` PR**，且合并后立即：
> ```bash
> git config blame.ignoreRevsFile .git-blame-ignore-revs
> git rev-parse HEAD >> .git-blame-ignore-revs
> ```
> 否则以后 `git blame` 每一行都指向"格式化提交"，追溯全废。

### 4.3 `config/checkstyle/checkstyle.xml`（可选，只留高价值规则）

```xml
<?xml version="1.0"?>
<!DOCTYPE module PUBLIC
    "-//Checkstyle//DTD Checkstyle Configuration 1.3//EN"
    "https://checkstyle.org/dtds/configuration_1_3.dtd">
<module name="Checker">
    <property name="charset" value="UTF-8"/>
    <property name="severity" value="error"/>

    <module name="TreeWalker">
        <!-- 未使用的 import：Spotless 已处理，这里兜底 -->
        <module name="UnusedImports"/>
        <!-- 空代码块（catch 里什么都不做是典型事故源） -->
        <module name="EmptyBlock">
            <property name="option" value="text"/>
            <property name="tokens"
                      value="LITERAL_CATCH,LITERAL_FINALLY,LITERAL_IF,LITERAL_FOR,LITERAL_WHILE"/>
        </module>
        <!-- 禁止空的 catch：必须至少记日志 -->
        <module name="IllegalCatch">
            <property name="illegalClassNames" value="java.lang.Throwable"/>
        </module>
        <!-- switch 必须有 default -->
        <module name="MissingSwitchDefault"/>
        <!-- equals 与 hashCode 必须成对重写 -->
        <module name="EqualsHashCode"/>
        <!-- 字符串比较不能用 == -->
        <module name="StringLiteralEquality"/>
        <!-- 禁止 System.out / printStackTrace -->
        <module name="RegexpSinglelineJava">
            <property name="format" value="System\.(out|err)\.print"/>
            <property name="message" value="生产代码禁止 System.out/err，请用 log 或 operLogService"/>
        </module>
        <module name="RegexpSinglelineJava">
            <property name="format" value="\.printStackTrace\(\)"/>
            <property name="message" value="禁止 printStackTrace，请抛 BizException 或 log.error"/>
        </module>
    </module>
</module>
```

接入：

```xml
<plugin>
    <groupId>org.apache.maven.plugins</groupId>
    <artifactId>maven-checkstyle-plugin</artifactId>
    <version>3.3.1</version>
    <configuration>
        <configLocation>config/checkstyle/checkstyle.xml</configLocation>
        <failOnViolation>true</failOnViolation>
        <violationSeverity>error</violationSeverity>
        <includeTestSourceDirectory>true</includeTestSourceDirectory>
    </configuration>
    <executions>
        <execution>
            <id>checkstyle</id>
            <phase>validate</phase>
            <goals><goal>check</goal></goals>
        </execution>
    </executions>
</plugin>
```

> 这些规则都是**真出过事**的类型：空 catch 吞异常、`==` 比字符串、`printStackTrace` 掩盖错误。它们不会引发风格争论。

## 五、CI 流水线

### 5.1 `.github/workflows/ci.yml`

```yaml
name: CI

on:
  pull_request:
    branches: [main]
  push:
    branches: [main]

jobs:
  backend:
    name: 后端 · 构建 + 检查 + 测试
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: backend
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'
          cache: maven

      - name: Spotless 格式检查
        run: mvn -B -s ../maven-settings.xml spotless:check

      - name: Checkstyle
        run: mvn -B -s ../maven-settings.xml checkstyle:check

      - name: 编译 + 测试
        run: mvn -B -s ../maven-settings.xml verify

      - name: 上传测试报告
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: backend-test-report
          path: |
            backend/target/surefire-reports/
            backend/target/sentiment-eval/
          retention-days: 7

  frontend:
    name: 前端 · Lint + 格式 + 构建
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: frontend
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: npm
          cache-dependency-path: frontend/package-lock.json

      - run: npm ci
      - run: npm run lint
      - run: npm run format:check
      - run: npm run build
```

## 六、本地提交前拦截（可选但强烈建议）

`.git/hooks/pre-commit`（`chmod +x`）：

```bash
#!/bin/sh
# 只检查本次暂存的文件，保证提交快（<5s）

staged_js=$(git diff --cached --name-only --diff-filter=ACM | grep -E '^frontend/src/.*\.(js|vue)$')
if [ -n "$staged_js" ]; then
  echo "→ ESLint 检查暂存文件..."
  (cd frontend && npx eslint $(echo "$staged_js" | sed 's|^frontend/||') ) || {
    echo "✗ ESLint 未通过，提交已中止"; exit 1; }
fi

staged_java=$(git diff --cached --name-only --diff-filter=ACM | grep -E '^backend/src/.*\.java$')
if [ -n "$staged_java" ]; then
  echo "→ 后端格式检查..."
  (cd backend && mvn -q -s ../maven-settings.xml spotless:check) || {
    echo "✗ 格式未通过，请先执行 mvn spotless:apply"; exit 1; }
fi

echo "✓ 提交前检查通过"
```

> **只检查暂存文件**是关键。全量检查会让每次提交等 30 秒，团队会本能地加 `--no-verify`，机制立刻失效。

## 七、schema 一致性守卫（对应红线 R8）

你们踩过 `resources/db/schema.sql` 与 `sql/schema.sql` 不同步导致启动失败的坑。用一个脚本守住：

`scripts/check-schema-sync.sh`

```bash
#!/bin/sh
# 校验两份 schema 是否一致；不一致则退出码 1
set -e
A="sql/schema.sql"
B="backend/src/main/resources/db/schema.sql"

if [ ! -f "$A" ] || [ ! -f "$B" ]; then
  echo "✗ 缺少 schema 文件：$A 或 $B"; exit 1
fi

if ! diff -q <(tr -d '\r' < "$A" | sed '/^\s*$/d') \
             <(tr -d '\r' < "$B" | sed '/^\s*$/d') > /dev/null; then
  echo "✗ 两份 schema 不一致，请统一后再提交："
  diff -u <(tr -d '\r' < "$A") <(tr -d '\r' < "$B") | head -40
  exit 1
fi

echo "✓ schema 一致"
```

接进 CI（后端 job 里加一步）：

```yaml
      - name: schema 一致性
        working-directory: .
        run: sh scripts/check-schema-sync.sh
```

**更彻底的做法**（推荐，消灭第二份副本）：在 `pom.xml` 里构建时复制，删除 `resources/db/schema.sql`：

```xml
<plugin>
    <groupId>org.apache.maven.plugins</groupId>
    <artifactId>maven-resources-plugin</artifactId>
    <executions>
        <execution>
            <id>copy-schema</id>
            <phase>process-resources</phase>
            <goals><goal>copy-resources</goal></goals>
            <configuration>
                <outputDirectory>${project.build.outputDirectory}/db</outputDirectory>
                <resources>
                    <resource>
                        <directory>${project.basedir}/../sql</directory>
                        <includes><include>schema.sql</include></includes>
                    </resource>
                </resources>
            </configuration>
        </execution>
    </executions>
</plugin>
```

## 九、一次性配置清单

按顺序执行，约 2 小时可全部落地：

```bash
# 1. 版本控制（P0-1）
cd "D:/桌面/Graduation project/mentor-comment-system"
git init && git branch -M main
# 写入 .gitignore → git add . && git commit -m "chore: 建立版本控制基线" && git tag baseline-v1

# 2. 前端工具链（P0-3）
cd frontend

# 写入 eslint.config.js / .prettierrc → 更新 package.json scripts
```

> 第 3 步的 `spotless:apply` 会改动大量文件。**单独提一个 PR**，不要和业务改动混在一起（见 `02-审查流程.md` §3.2）。
